cmake_minimum_required(VERSION 3.25)
project(MIGsdl2Example LANGUAGES CXX)
include(support/sdk.cmake)
find_package(SDL2 CONFIG REQUIRED)
find_package(SDL2_ttf CONFIG QUIET)
if(NOT TARGET SDL2_ttf::SDL2_ttf)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
    add_library(SDL2_ttf::SDL2_ttf ALIAS PkgConfig::SDL_TTF)
endif()
foreach(target mig-sdl2 mig-sdl2-profile)
    add_executable(${target} main.cpp application.cpp)
    target_link_libraries(${target} PRIVATE MIG::native MIG::format MIG::hands
        SDL2::SDL2 SDL2_ttf::SDL2_ttf)
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(TARGET SDL2::SDL2main)
        target_link_libraries(${target} PRIVATE SDL2::SDL2main)
    endif()
    mig_example_assets(${target})
endforeach()
