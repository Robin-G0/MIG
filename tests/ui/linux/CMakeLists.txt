find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
foreach(application configurator controller)
    if(TARGET mig-${application})
        target_sources(mig-${application} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${application}_ui_tests.cpp")
        target_include_directories(mig-${application} PRIVATE "${PROJECT_SOURCE_DIR}/src/linux-apps")
        target_compile_definitions(mig-${application} PRIVATE MIG_APP_TESTS)
        add_test(NAME linux-${application}-ui COMMAND mig-${application} --ui-test)
        set_tests_properties(linux-${application}-ui PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30 RUN_SERIAL TRUE)
    endif()
endforeach()
add_executable(mig-qt-theme-tests qt_theme_tests.cpp "${PROJECT_SOURCE_DIR}/src/linux-apps/theme.cpp")
target_link_libraries(mig-qt-theme-tests PRIVATE Qt6::Widgets)
add_test(NAME qt-theme COMMAND mig-qt-theme-tests)
set_tests_properties(qt-theme PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
